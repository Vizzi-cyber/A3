## 2.1 OLED显示屏(I2C)

### 实验目的
掌握 I2C 通信，驱动 OLED 显示屏。

### 硬件连接
```
STM32          OLED模块
PB10(SCL) ──── SCL
PB11(SDA) ──── SDA
3.3V ────────── VCC
GND ────────── GND
```

**接线图参考**: `1-1 接线图/4-1 OLED显示屏.jpg`

### 知识点：I2C协议

#### I2C时序
```
起始信号:  SCL高电平时，SDA由高变低
停止信号:  SCL高电平时，SDA由低变高
发送数据:  SCL低电平时准备数据，SCL高电平时读取数据
应答信号:  第9个时钟周期，SDA被拉低表示ACK
```

#### I2C设备地址
```
SSD1306 OLED: 0x78 (写) / 0x79 (读)
MPU6050:      0xD0 (写) / 0xD1 (读)
```

### OLED驱动原理

#### SSD1306初始化序列
```c
void OLED_Init(void)
{
    // 上电延时
    Delay_ms(100);
    
    // 发送初始化命令序列
    OLED_WriteCmd(0xAE);  // 关闭显示
    OLED_WriteCmd(0x20);  // 内存寻址模式
    OLED_WriteCmd(0x10);  // 页地址模式
    OLED_WriteCmd(0xB0);  // 起始页0
    OLED_WriteCmd(0xC8);  // COM扫描方向
    OLED_WriteCmd(0x00);  // 列地址低4位
    OLED_WriteCmd(0x10);  // 列地址高4位
    OLED_WriteCmd(0x40);  // 起始行
    OLED_WriteCmd(0x81);  // 对比度
    OLED_WriteCmd(0xCF);
    OLED_WriteCmd(0xA1);  // 段重映射
    OLED_WriteCmd(0xA6);  // 正常显示
    OLED_WriteCmd(0xA8);  // 多路复用率
    OLED_WriteCmd(0x3F);
    OLED_WriteCmd(0xA4);  // 全局显示开启
    OLED_WriteCmd(0xD3);  // 显示偏移
    OLED_WriteCmd(0x00);
    OLED_WriteCmd(0xD5);  // 时钟分频
    OLED_WriteCmd(0xF0);
    OLED_WriteCmd(0xD9);  // 预充电期
    OLED_WriteCmd(0x22);
    OLED_WriteCmd(0xDA);  // COM引脚配置
    OLED_WriteCmd(0x12);
    OLED_WriteCmd(0xDB);  // VCOMH电压
    OLED_WriteCmd(0x20);
    OLED_WriteCmd(0x8D);  // 电荷泵
    OLED_WriteCmd(0x14);
    OLED_WriteCmd(0xAF);  // 开启显示
}
```

### 完整代码

#### OLED.c
```c
#include "stm32f10x.h"
#include "Delay.h"

#define OLED_SCL_GPIO    GPIOB
#define OLED_SCL_PIN     GPIO_Pin_10
#define OLED_SDA_GPIO    GPIOB
#define OLED_SDA_PIN     GPIO_Pin_11

#define OLED_SCL_1()     GPIO_SetBits(OLED_SCL_GPIO, OLED_SCL_PIN)
#define OLED_SCL_0()     GPIO_ResetBits(OLED_SCL_GPIO, OLED_SCL_PIN)
#define OLED_SDA_1()     GPIO_SetBits(OLED_SDA_GPIO, OLED_SDA_PIN)
#define OLED_SDA_0()     GPIO_ResetBits(OLED_SDA_GPIO, OLED_SDA_PIN)

// 软件I2C起始信号
void OLED_Start(void)
{
    OLED_SDA_1();
    OLED_SCL_1();
    OLED_SDA_0();
    OLED_SCL_0();
}

// 软件I2C停止信号
void OLED_Stop(void)
{
    OLED_SDA_0();
    OLED_SCL_1();
    OLED_SDA_1();
}

// 软件I2C发送一个字节
void OLED_WriteByte(uint8_t data)
{
    for (int i = 0; i < 8; i++)
    {
        if (data & 0x80)
            OLED_SDA_1();
        else
            OLED_SDA_0();
        
        OLED_SCL_1();
        OLED_SCL_0();
        data <<= 1;
    }
    // 等待应答
    OLED_SDA_1();
    OLED_SCL_1();
    OLED_SCL_0();
}

// 发送命令
void OLED_WriteCmd(uint8_t cmd)
{
    OLED_Start();
    OLED_WriteByte(0x78);  // 地址
    OLED_WriteByte(0x00);  // 命令
    OLED_WriteByte(cmd);
    OLED_Stop();
}

// 发送数据
void OLED_WriteData(uint8_t data)
{
    OLED_Start();
    OLED_WriteByte(0x78);  // 地址
    OLED_WriteByte(0x40);  // 数据
    OLED_WriteByte(data);
    OLED_Stop();
}

// 设置光标位置
void OLED_SetPos(uint8_t x, uint8_t y)
{
    OLED_WriteCmd(0xB0 + y);
    OLED_WriteCmd(((x & 0xF0) >> 4) | 0x10);
    OLED_WriteCmd(x & 0x0F);
}

// 清屏
void OLED_Clear(void)
{
    for (int y = 0; y < 8; y++)
    {
        OLED_WriteCmd(0xB0 + y);
        OLED_WriteCmd(0x00);
        OLED_WriteCmd(0x10);
        for (int x = 0; x < 128; x++)
        {
            OLED_WriteData(0x00);
        }
    }
}

// 显示字符（6x8字体）
void OLED_ShowChar(uint8_t x, uint8_t y, char ch)
{
    // 字符库偏移（空格的ASCII码为32）
    ch = ch - ' ';
    OLED_SetPos(x, y);
    for (int i = 0; i < 6; i++)
    {
        OLED_WriteData(OLED_F6x8[ch][i]);
    }
}

// 显示字符串
void OLED_ShowString(uint8_t x, uint8_t y, char *str)
{
    while (*str)
    {
        OLED_ShowChar(x, y, *str);
        x += 6;
        str++;
    }
}
```

#### main.c
```c
#include "stm32f10x.h"
#include "Delay.h"
#include "OLED.h"

int main(void)
{
    OLED_Init();
    OLED_Clear();
    
    OLED_ShowString(0, 0, "Hello STM32!");
    OLED_ShowString(0, 2, "OLED Display");
    OLED_ShowString(0, 4, "I2C Mode");
    
    while (1)
    {
        
    }
}
```

## 2.2 OLED显示实验数据

### 实验目的
掌握 OLED 动态显示数据。

### 完整代码
```c
#include "stm32f10x.h"
#include "Delay.h"
#include "OLED.h"

int main(void)
{
    OLED_Init();
    OLED_Clear();
    
    uint16_t count = 0;
    char buffer[20];
    
    while (1)
    {
        count++;
        
        // 显示计数值
        sprintf(buffer, "Count: %d", count);
        OLED_ShowString(0, 0, buffer);
        
        // 显示时间
        uint32_t seconds = count / 10;
        sprintf(buffer, "Time: %ds", seconds);
        OLED_ShowString(0, 2, buffer);
        
        Delay_ms(100);
    }
}
```
