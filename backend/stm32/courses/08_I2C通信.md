## 8.1 软件I2C读写MPU6050

### 实验目的
软件模拟 I2C 时序，读取 MPU6050 数据。

### 硬件连接
```
STM32          MPU6050模块
PB10(SCL) ──── SCL
PB11(SDA) ──── SDA
3.3V ────────── VCC
GND ────────── GND
```

**接线图参考**: `1-1 接线图/10-1 软件I2C读写MPU6050.jpg`

### 知识点：MPU6050

#### MPU6050寄存器
```
0x6B: PWR_MGMT_1 (电源管理)
0x3B~0x40: 加速度计数据 (XH,XL,YH,YL,ZH,ZL)
0x43~0x48: 陀螺仪数据
0x75: WHO_AM_I (设备ID, 应为0x68)
```

### 完整代码
```c
#include "stm32f10x.h"
#include "Delay.h"
#include "OLED.h"

#define MPU6050_ADDR  0xD0

void I2C_Init(void)
{
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOB, ENABLE);
    
    GPIO_InitTypeDef GPIO_InitStructure;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_OD;    // 开漏输出
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_10 | GPIO_Pin_11;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOB, &GPIO_InitStructure);
    
    GPIO_SetBits(GPIOB, GPIO_Pin_10);  // SCL高
    GPIO_SetBits(GPIOB, GPIO_Pin_11);  // SDA高
}

// 起始信号
void I2C_Start(void)
{
    SDA_1();
    SCL_1();
    SDA_0();
    SCL_0();
}

// 停止信号
void I2C_Stop(void)
{
    SDA_0();
    SCL_1();
    SDA_1();
}

// 发送一个字节
void I2C_SendByte(uint8_t data)
{
    for (int i = 0; i < 8; i++)
    {
        if (data & 0x80)
            SDA_1();
        else
            SDA_0();
        
        SCL_1();
        SCL_0();
        data <<= 1;
    }
    SDA_1();
    SCL_1();
    SCL_0();
}

// 读取一个字节
uint8_t I2C_ReceiveByte(void)
{
    uint8_t data = 0;
    SDA_1();
    
    for (int i = 0; i < 8; i++)
    {
        SCL_1();
        data <<= 1;
        if (SDA_READ())
            data |= 0x01;
        SCL_0();
    }
    return data;
}

// 写寄存器
void MPU6050_WriteReg(uint8_t reg, uint8_t data)
{
    I2C_Start();
    I2C_SendByte(MPU6050_ADDR);
    I2C_SendByte(reg);
    I2C_SendByte(data);
    I2C_Stop();
}

// 读寄存器
uint8_t MPU6050_ReadReg(uint8_t reg)
{
    uint8_t data;
    I2C_Start();
    I2C_SendByte(MPU6050_ADDR);
    I2C_SendByte(reg);
    I2C_Start();
    I2C_SendByte(MPU6050_ADDR | 0x01);
    data = I2C_ReceiveByte();
    I2C_Stop();
    return data;
}

// 初始化MPU6050
void MPU6050_Init(void)
{
    MPU6050_WriteReg(0x6B, 0x80);  // 复位
    Delay_ms(100);
    MPU6050_WriteReg(0x6B, 0x00);  // 唤醒
    MPU6050_WriteReg(0x1B, 0x00);  // 陀螺仪±250°/s
    MPU6050_WriteReg(0x1C, 0x00);  // 加速度计±2g
}

// 读取加速度
void MPU6050_ReadAccel(int16_t *x, int16_t *y, int16_t *z)
{
    uint8_t buf[6];
    buf[0] = MPU6050_ReadReg(0x3B);
    buf[1] = MPU6050_ReadReg(0x3C);
    buf[2] = MPU6050_ReadReg(0x3D);
    buf[3] = MPU6050_ReadReg(0x3E);
    buf[4] = MPU6050_ReadReg(0x3F);
    buf[5] = MPU6050_ReadReg(0x40);
    
    *x = (buf[0] << 8) | buf[1];
    *y = (buf[2] << 8) | buf[3];
    *z = (buf[4] << 8) | buf[5];
}

int main(void)
{
    OLED_Init();
    I2C_Init();
    MPU6050_Init();
    
    int16_t ax, ay, az;
    char buffer[20];
    
    while (1)
    {
        MPU6050_ReadAccel(&ax, &ay, &az);
        
        sprintf(buffer, "X:%6d", ax);
        OLED_ShowString(0, 0, buffer);
        sprintf(buffer, "Y:%6d", ay);
        OLED_ShowString(0, 2, buffer);
        sprintf(buffer, "Z:%6d", az);
        OLED_ShowString(0, 4, buffer);
        
        Delay_ms(100);
    }
}
```
