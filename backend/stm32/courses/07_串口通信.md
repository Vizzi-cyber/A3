## 7.1 串口发送

### 实验目的
掌握 UART 配置，实现串口数据发送。

### 硬件连接
```
STM32          USB-TTL模块
PA9(TX) ──────── RXD
PA10(RX) ─────── TXD
GND ───────────── GND
```

**接线图参考**: `1-1 接线图/9-1 串口发送.jpg`

### 知识点：UART参数
```
波特率: 9600, 115200 (常用)
数据位: 8位
停止位: 1位
校验位: 无
```

### 完整代码
```c
#include "stm32f10x.h"
#include <stdio.h>

void USART_Init(void)
{
    // 1. 开启时钟
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_USART1, ENABLE);
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);
    
    // 2. 配置PA9(TX)为复用推挽输出
    GPIO_InitTypeDef GPIO_InitStructure;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_9;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 3. 配置PA10(RX)为浮空输入
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_10;
    GPIO_Init(GPIOA, &GPIO_InitStructure);
    
    // 4. 配置USART1
    USART_InitTypeDef USART_InitStructure;
    USART_InitStructure.USART_BaudRate = 115200;
    USART_InitStructure.USART_WordLength = USART_WordLength_8b;
    USART_InitStructure.USART_StopBits = USART_StopBits_1;
    USART_InitStructure.USART_Parity = USART_Parity_No;
    USART_InitStructure.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    USART_InitStructure.USART_Mode = USART_Mode_Tx | USART_Mode_Rx;
    USART_Init(USART1, &USART_InitStructure);
    
    USART_Cmd(USART1, ENABLE);
}

// 发送一个字节
void USART_SendByte(uint8_t byte)
{
    USART_SendData(USART1, byte);
    while (USART_GetFlagStatus(USART1, USART_FLAG_TXE) == RESET);
}

// 发送字符串
void USART_SendString(char *str)
{
    while (*str)
    {
        USART_SendByte(*str);
        str++;
    }
}

// 重定向printf到串口
int fputc(int ch, FILE *f)
{
    USART_SendByte(ch);
    return ch;
}

int main(void)
{
    USART_Init();
    
    printf("Hello STM32!\r\n");
    printf("UART Test\r\n");
    
    uint32_t count = 0;
    
    while (1)
    {
        printf("Count: %lu\r\n", count++);
        for (volatile int i = 0; i < 500000; i++);
    }
}
```

### 实验现象
串口调试助手收到 STM32 发送的数据。

## 7.2 串口发送+接收

### 实验目的
实现串口双向通信。

### 完整代码
```c
#include "stm32f10x.h"

void USART_Init(void)
{
    // ... 同上
    
    // 使能接收中断
    USART_ITConfig(USART1, USART_IT_RXNE, ENABLE);
    
    NVIC_InitTypeDef NVIC_InitStructure;
    NVIC_InitStructure.NVIC_IRQChannel = USART1_IRQn;
    NVIC_InitStructure.NVIC_IRQChannelPreemptionPriority = 2;
    NVIC_InitStructure.NVIC_IRQChannelSubPriority = 0;
    NVIC_InitStructure.NVIC_IRQChannelCmd = ENABLE;
    NVIC_Init(&NVIC_InitStructure);
    
    USART_Cmd(USART1, ENABLE);
}

// 接收缓冲区
volatile uint8_t rxData;
volatile uint8_t rxBuffer[256];
volatile uint16_t rxIndex = 0;

void USART1_IRQHandler(void)
{
    if (USART_GetITStatus(USART1, USART_IT_RXNE) != RESET)
    {
        rxData = USART_ReceiveData(USART1);
        rxBuffer[rxIndex++] = rxData;
        
        // 回显
        USART_SendData(USART1, rxData);
    }
}

int main(void)
{
    USART_Init();
    printf("UART Echo Ready\r\n");
    
    while (1)
    {
        // 主循环可以处理接收的数据
    }
}
```

### 实验现象
串口发送什么，STM32 就回显什么。

## 7.3 串口收发HEX数据包

### 实验目的
掌握二进制数据包的收发。

### 数据帧格式
```
帧头(2字节) | 命令(1字节) | 长度(1字节) | 数据(N字节) | 校验(1字节)
0xAA 0x55    CMD          LEN          DATA...       SUM
```

### 完整代码
```c
#include "stm32f10x.h"

typedef struct {
    uint8_t header[2];   // 帧头 0xAA 0x55
    uint8_t command;     // 命令
    uint8_t length;      // 数据长度
    uint8_t data[256];   // 数据
    uint8_t checksum;    // 校验和
} Packet_t;

volatile Packet_t rxPacket;
volatile uint8_t rxState = 0;
volatile uint8_t rxDataIndex = 0;

void USART1_IRQHandler(void)
{
    if (USART_GetITStatus(USART1, USART_IT_RXNE) != RESET)
    {
        uint8_t data = USART_ReceiveData(USART1);
        
        switch (rxState)
        {
            case 0:  // 等待帧头0xAA
                if (data == 0xAA) rxState = 1;
                break;
            case 1:  // 等待帧头0x55
                if (data == 0x55) rxState = 2;
                else rxState = 0;
                break;
            case 2:  // 接收命令
                rxPacket.command = data;
                rxState = 3;
                break;
            case 3:  // 接收长度
                rxPacket.length = data;
                rxDataIndex = 0;
                rxState = 4;
                break;
            case 4:  // 接收数据
                rxPacket.data[rxDataIndex++] = data;
                if (rxDataIndex >= rxPacket.length)
                    rxState = 5;
                break;
            case 5:  // 接收校验和
                rxPacket.checksum = data;
                // 验证校验和
                // 处理数据包
                rxState = 0;
                break;
        }
    }
}

// 发送数据包
void USART_SendPacket(uint8_t cmd, uint8_t *data, uint8_t len)
{
    uint8_t checksum = 0;
    
    USART_SendByte(0xAA);
    USART_SendByte(0x55);
    USART_SendByte(cmd);
    USART_SendByte(len);
    
    for (int i = 0; i < len; i++)
    {
        USART_SendByte(data[i]);
        checksum += data[i];
    }
    
    USART_SendByte(checksum);
}

int main(void)
{
    USART_Init();
    printf("HEX Packet Ready\r\n");
    
    while (1)
    {
        
    }
}
```

## 7.4 串口收发文本数据包

### 实验目的
掌握文本命令解析。

### 命令格式
```
SET LED 1\r\n      - 开启LED
SET LED 0\r\n      - 关闭LED
GET ADC\r\n        - 读取ADC值
PING\r\n           - 测试连接
```

### 完整代码
```c
#include "stm32f10x.h"
#include <string.h>
#include <stdlib.h>

#define RX_BUF_SIZE 128

volatile char rxBuffer[RX_BUF_SIZE];
volatile uint16_t rxCount = 0;
volatile uint8_t rxComplete = 0;

void USART1_IRQHandler(void)
{
    if (USART_GetITStatus(USART1, USART_IT_RXNE) != RESET)
    {
        char data = USART_ReceiveData(USART1);
        
        if (data == '\n' || data == '\r')
        {
            if (rxCount > 0)
            {
                rxBuffer[rxCount] = '\0';
                rxComplete = 1;
            }
        }
        else
        {
            if (rxCount < RX_BUF_SIZE - 1)
            {
                rxBuffer[rxCount++] = data;
            }
        }
    }
}

// 解析命令
void ParseCommand(char *cmd)
{
    char *token;
    
    token = strtok(cmd, " ");
    
    if (strcmp(token, "SET") == 0)
    {
        token = strtok(NULL, " ");
        if (strcmp(token, "LED") == 0)
        {
            token = strtok(NULL, " ");
            if (strcmp(token, "1") == 0)
            {
                GPIO_SetBits(GPIOA, GPIO_Pin_0);
                printf("LED ON\r\n");
            }
            else if (strcmp(token, "0") == 0)
            {
                GPIO_ResetBits(GPIOA, GPIO_Pin_0);
                printf("LED OFF\r\n");
            }
        }
    }
    else if (strcmp(token, "GET") == 0)
    {
        token = strtok(NULL, " ");
        if (strcmp(token, "ADC") == 0)
        {
            uint16_t adcValue = ADC_Read();
            printf("ADC: %d\r\n", adcValue);
        }
    }
    else if (strcmp(token, "PING") == 0)
    {
        printf("PONG\r\n");
    }
}

int main(void)
{
    USART_Init();
    printf("Text Command Ready\r\n");
    
    while (1)
    {
        if (rxComplete)
        {
            rxComplete = 0;
            ParseCommand((char *)rxBuffer);
            rxCount = 0;
        }
    }
}
```
